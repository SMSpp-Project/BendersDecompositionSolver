# A txt description of a BlockSolverConfig registering two Solver on the same
# Block, which is what a cross-check is: a BendersDecompositionSolver in the
# MILP regime, giving the Block back as it was at the end of each compute(),
# and a :MILPSolver reading the whole tree, i.e., the extensive form of the
# same problem. The two describe one problem, hence they must agree, and the
# second one is right only because the first gives back the complicating
# Variable it takes out of the Constraint of the subproblems.
#
#   Antonio Frangioni
#   Donato Meoli
#
# - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
# - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -

BlockSolverConfig     # exact type of the Configuration object

0  # the BlockSolverConfig is not a "differential" one

2  # number of (the names of) Solver in this BlockSolverConfig
# now all the names of the Solver - - - - - - - - - - - - - - - - - - - - - -
BendersDecompositionSolver # name of 1st Solver
GRBMILPSolver              # name of 2nd Solver

2  # number of ComputeConfig in this BlockSolverConfig

# now all the ComputeConfig
# 1st ComputeConfig - - - - - - - - - - - - - - - - - - - - - - - - - - - - -

ComputeConfig # exact type of the ComputeConfig object

0  # f_diff == 0 ==> all non-provided parameters are set to the default value
   # f_diff == 1 ==> all non-provided parameters are not changed

2  # number of integer parameters

# now all the integer parameters
int_BDSlv_Regime  1  # 0 = convex master, 1 = MILP master
int_BDSlv_Restore 1  # 0 = the reformulation stays, 1 = the Block is given
                     #     back after each compute()

0  # number of double parameters

# now all the double parameters
# [none]

2  # number of string parameters

# now all the string parameters
str_Bsub_BSCfg  BSPar_sub.txt     # BlockSolverConfig of the subproblems
str_Mstr_BSCfg  BSPar_master_milp.txt # BlockSolverConfig of the master

0  # number of vector-of-int parameters

# now all the vector-of-int parameters
# [none]

0  # number of vector-of-double parameters

# now all the vector-of-double parameters
# [none]

0  # number of vector-of-string parameters

# now all the vector-of-string parameters
# [none]

# pointer to the "extra" Configuration
* # [none]

# end of 1st ComputeConfig - - - - - - - - - - - - - - - - - - - - - - - - - -

# 2nd ComputeConfig - - - - - - - - - - - - - - - - - - - - - - - - - - - - -

ComputeConfig # exact type of the ComputeConfig object

0  # f_diff == 0 ==> all non-provided parameters are set to the default value
   # f_diff == 1 ==> all non-provided parameters are not changed

3  # number of integer parameters

# now all the integer parameters
intLogVerb        0  # LogVerb, log verbosity of the MILPSolver
Threads           1  # one thread, so that the time is that of a sequential
                     # algorithm, as every other one here is
intRelaxIntVars   1  # solve the continuous (LP) relaxation

1  # number of double parameters

# now all the double parameters
dblRelAcc      1e-9  # relative accuracy required to solution

0  # number of string parameters

# now all the string parameters
# [none]

0  # number of vector-of-int parameters

# now all the vector-of-int parameters
# [none]

0  # number of vector-of-double parameters

# now all the vector-of-double parameters
# [none]

0  # number of vector-of-string parameters
